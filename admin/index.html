async function publishPost(isDraft = false) { const title = document.getElementById('postTitle').value.trim(); const slug = document.getElementById('postSlug').value.trim() || title.toLowerCase().replace(/[^a-z0-9]+/g, '-'); const desc = document.getElementById('metaDesc').value.trim(); const content = document.getElementById('postContent').value.trim(); const msgEl = document.getElementById('actionMsg'); if(!title || !content) { msgEl.style.color = "var(--danger)"; msgEl.innerText = "Please fill in title and content!"; return; } let fileNameOnly = isDraft ? `draft-${slug}.md` : `${new Date().toISOString().split('T')[0]}-${slug}.md`; let fileContent = `---\ntitle: "${title}"\ndescription: "${desc}"\ndraft: ${isDraft}\n---\n\n${content}`; let encodedContent = btoa(unescape(encodeURIComponent(fileContent))); msgEl.style.color = "var(--text-muted)"; msgEl.innerText = "Saving to GitHub..."; try { let checkRes = await fetch(`https://api.github.com/repos/${REPO}/contents/${fileNameOnly}`, { headers: { 'Authorization': `Bearer ${authData.token}` } }); let sha = null; if(checkRes.ok) { let fileData = await checkRes.json(); sha = fileData.sha; } let bodyData = { message: `CMS Update: ${title}`, content: encodedContent }; if(sha) bodyData.sha = sha; let res = await fetch(`https://api.github.com/repos/${REPO}/contents/${fileNameOnly}`, { method: 'PUT', headers: { 'Authorization': `Bearer ${authData.token}`, 'Content-Type': 'application/json' }, body: JSON.stringify(bodyData) }); if(res.ok) { msgEl.style.color = "var(--success)"; msgEl.innerText = "Successfully saved!"; setTimeout(() => { location.reload(); }, 1200); } else { let err = await res.json(); throw new Error(err.message || "Failed"); } } catch(e) { msgEl.style.color = "var(--danger)"; alert("Error: " + e.message); } }
